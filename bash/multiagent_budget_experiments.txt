# Budgeted Coordination Law: synthetic-DAG multi-agent experiments
# Run from the MASModel repository root: bash multiagent_budget_experiments.txt
#
# IMPORTANT: agent.total_decision_budget is a SYSTEM worker budget. The code
# assigns floor(total_budget / n_base_agents) calls to each worker. The primary
# sweep therefore uses total_budget=n*per_agent_budget when local budget is
# the controlled variable.

set -euo pipefail

DATASET="datasets/threshold_families/threshold-3-5-7-spread.json"
MODEL="openai/qwen-flash"
COMMON=(
  dataset=synthetic-dag
  dataset.local_path="$DATASET"
  llm.model="$MODEL"
  llm.params.temperature=0.0
  max_instances=100
  num_workers=3
  debug=false
  log_langfuse=false
  use_disk_cache=true
  agent.min_iterations_per_agent=0
  agent.use_remaining_budget_for_verification=true
)

run_independent() {
  local n="$1"; local budget="$2"
  python scripts/run_experiment.py \
    agent=multi-agent-independent "${COMMON[@]}" \
    agent.n_base_agents="$n" agent.total_decision_budget="$budget"
}

run_coordinated() {
  local arch="$1"; local n="$2"; local budget="$3"; local rounds="$4"
  python scripts/run_experiment.py \
    agent="$arch" "${COMMON[@]}" \
    agent.n_base_agents="$n" agent.total_decision_budget="$budget" \
    agent.max_rounds="$rounds"
}


# PHASE 1: fixed per-agent lifetime budget. b is local budget, n is team size,
# so system budget is exactly n*b. Main architectures: Independent, Centralized,
# Decentralized. n={2,3,4,5}, b={3,5,7,9,11}; 60 runs.
for n in 2 3 4 5; do
  for b in 3 5 7 9 11; do
    run_independent "$n" "$((n * b))"
  done
done
for arch in multi-agent-centralized multi-agent-decentralized; do
  for n in 2 3 4 5; do
    for b in 3 5 7 9 11; do
      run_coordinated "$arch" "$n" "$((n * b))" 3
    done
  done
done

# PHASE 1B: limited Hybrid confirmation at n=3; 3 runs.
for b in 3 7 11; do
  run_coordinated multi-agent-hybrid 3 "$((3 * b))" 3
done

# PHASE 2: fixed SYSTEM budget. This separates coordination/parallelism from
# simply buying more calls. Use four evenly spaced total budgets
# B={16,28,40,52}. For n={2,3,4,5}, nominal per-agent budgets are:
#   B=16: {8,5,4,3}, remainders {0,1,0,1}
#   B=28: {14,9,7,5}, remainders {0,1,0,3}
#   B=40: {20,13,10,8}, remainders {0,1,0,0}
#   B=52: {26,17,13,10}, remainders {0,1,0,2}
# Thus n=2 and n=4 are exact; n=3 loses one call; and n=5 loses at most
# three calls. The unallocated remainder is 10 calls per architecture across
# the 16 budget/team-size cells (30 calls across the three core architectures).
# Core architectures only, 48 runs.
for arch in multi-agent-independent multi-agent-centralized multi-agent-decentralized; do
  for total_budget in 16 28 40 52; do
    for n in 2 3 4 5; do
      if [[ "$arch" == "multi-agent-independent" ]]; then
        run_independent "$n" "$total_budget"
      else
        run_coordinated "$arch" "$n" "$total_budget" 3
      fi
    done
  done
done

# PHASE 2B: Hybrid fixed-system-budget confirmation at n=5; 4 runs. This
# explicitly checks the largest team size under the same total budgets.
for total_budget in 16 28 40 52; do
  run_coordinated multi-agent-hybrid 5 "$total_budget" 3
done

# A canonical n=3 Hybrid reference for the middle total budget; 1 run.
for total_budget in 40; do
  run_coordinated multi-agent-hybrid 3 "$total_budget" 3
done

# PHASE 3: coordination-horizon ablation. n=3, system budget=21 (local=7),
# max_rounds={1,3,5}; 3 architectures x 3 horizons = 9 runs.
for arch in multi-agent-centralized multi-agent-decentralized multi-agent-hybrid; do
  for rounds in 1 3 5; do
    run_coordinated "$arch" 3 21 "$rounds"
  done
done

# PHASE 4: verification ablation. n=3, system budget=21, max_rounds=3;
# compare deferred verification on/off for three coordinated architectures;
# 6 runs.
for arch in multi-agent-centralized multi-agent-decentralized multi-agent-hybrid; do
  python scripts/run_experiment.py agent="$arch" "${COMMON[@]}" agent.n_base_agents=3 agent.total_decision_budget=21 agent.max_rounds=3 agent.use_remaining_budget_for_verification=true
  python scripts/run_experiment.py agent="$arch" "${COMMON[@]}" agent.n_base_agents=3 agent.total_decision_budget=21 agent.max_rounds=3 agent.use_remaining_budget_for_verification=false
done

# PHASE 5: arithmetic difficulty control. The three threshold-5 files preserve
# the structural threshold while changing arithmetic difficulty. n=3, local=7,
# core architectures; 9 runs.
for difficulty_dataset in datasets/threshold_families/threshold-5-easy.json datasets/threshold_families/threshold-5-medium-paired.json datasets/threshold_families/threshold-5-hard-paired.json; do
  for arch in multi-agent-independent multi-agent-centralized multi-agent-decentralized; do
    if [[ "$arch" == "multi-agent-independent" ]]; then
      python scripts/run_experiment.py agent="$arch" dataset=synthetic-dag dataset.local_path="$difficulty_dataset" llm.model="$MODEL" llm.params.temperature=0.0 max_instances=100 num_workers=3 debug=false log_langfuse=false use_disk_cache=true agent.n_base_agents=3 agent.total_decision_budget=21 agent.min_iterations_per_agent=0 agent.use_remaining_budget_for_verification=true
    else
      python scripts/run_experiment.py agent="$arch" dataset=synthetic-dag dataset.local_path="$difficulty_dataset" llm.model="$MODEL" llm.params.temperature=0.0 max_instances=100 num_workers=3 debug=false log_langfuse=false use_disk_cache=true agent.n_base_agents=3 agent.total_decision_budget=21 agent.max_rounds=3 agent.min_iterations_per_agent=0 agent.use_remaining_budget_for_verification=true
    fi
  done
done

# Minimum paper-quality set: Phases 1, 2, and 3. Phases 4 and 5 are controls.
